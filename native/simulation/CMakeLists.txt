cmake_minimum_required(VERSION 3.21)
project(cyancular_simulation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE PATH "" FORCE)
add_subdirectory("${GODOT_CPP_DIR}" godot-cpp)
if(MINGW)
    target_compile_options(godot-cpp PRIVATE -include cstdlib)
endif()
add_library(cyancular_simulation SHARED simulation.cpp actor_body.cpp motion_track.cpp bullet_body.cpp weapon_body.cpp projectile_entry.cpp projectile_snapshot.cpp combat_kernel.cpp prediction_history.cpp terrain.cpp terrain_query.cpp performance_stats.cpp)
target_link_libraries(cyancular_simulation PRIVATE godot::cpp)
if(MSVC)
    target_compile_options(cyancular_simulation PRIVATE /fp:strict)
else()
    target_compile_options(cyancular_simulation PRIVATE -ffp-contract=off -fno-fast-math)
endif()
if(MINGW)
    target_link_options(cyancular_simulation PRIVATE -static)
endif()
set_target_properties(cyancular_simulation PROPERTIES PREFIX ""
    OUTPUT_NAME "simulation.${SIM_PLATFORM}.${SIM_ARCH}"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
