cmake_minimum_required(VERSION 3.21)
project(cyancular_native LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GODOTCPP_TARGET template_release CACHE STRING "")
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "")
set(BUILD_SHARED_LIB OFF CACHE BOOL "")
set(BUILD_STATIC_LIB ON CACHE BOOL "")
set(BUILD_EXAMPLES OFF CACHE BOOL "")
set(BUILD_TESTS OFF CACHE BOOL "")
set(BUILD_TOOLS OFF CACHE BOOL "")
set(ENABLE_ICE OFF CACHE BOOL "")
set(USE_STEAMWEBRTC OFF CACHE BOOL "")
set(Protobuf_USE_STATIC_LIBS ON)
if(WIN32)
    set(USE_CRYPTO BCrypt CACHE STRING "")
    set(PROCESS_GROUP_SOURCE process_group/process_group.cpp)
else()
    set(USE_CRYPTO OpenSSL CACHE STRING "")
    set(OPENSSL_USE_STATIC_LIBS TRUE)
    set(PROCESS_GROUP_SOURCE process_group/process_group_macos.cpp)
endif()

add_subdirectory("${GODOT_CPP_DIR}" godot-cpp)
add_subdirectory("${GNS_DIR}" gns)
find_package(Threads REQUIRED)

file(GLOB FAST_LZMA2_SOURCES "${FAST_LZMA2_DIR}/*.c")
list(REMOVE_ITEM FAST_LZMA2_SOURCES "${FAST_LZMA2_DIR}/xxhash.c")
add_library(fast_lzma2 STATIC ${FAST_LZMA2_SOURCES})
target_include_directories(fast_lzma2 PUBLIC "${FAST_LZMA2_DIR}")
target_compile_definitions(fast_lzma2 PUBLIC NO_XXHASH)
target_link_libraries(fast_lzma2 PRIVATE Threads::Threads)

add_library(cyr_gns SHARED gns/gns.cpp)
target_link_libraries(cyr_gns PRIVATE godot::cpp GameNetworkingSockets::static fast_lzma2)
set_target_properties(cyr_gns PROPERTIES PREFIX "" OUTPUT_NAME gns)

add_library(process_group SHARED "${PROCESS_GROUP_SOURCE}")
target_link_libraries(process_group PRIVATE godot::cpp)
set_target_properties(process_group PROPERTIES PREFIX "")
